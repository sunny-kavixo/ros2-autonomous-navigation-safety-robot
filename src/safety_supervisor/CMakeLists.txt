cmake_minimum_required(VERSION 3.8)
project(safety_supervisor)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(safety_policy src/safety_policy.cpp)
target_include_directories(safety_policy PUBLIC include)
target_compile_features(safety_policy PUBLIC cxx_std_17)
add_executable(safety_supervisor_node src/safety_supervisor_node.cpp)
target_link_libraries(safety_supervisor_node safety_policy)
target_include_directories(safety_supervisor_node PRIVATE include)
ament_target_dependencies(safety_supervisor_node rclcpp geometry_msgs sensor_msgs)
install(TARGETS safety_policy safety_supervisor_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/safety_supervisor)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_safety_policy test/test_safety_policy.cpp)
  target_link_libraries(test_safety_policy safety_policy)
  target_include_directories(test_safety_policy PRIVATE include)
endif()
ament_package()
